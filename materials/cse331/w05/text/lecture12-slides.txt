JSON

88private static String json(List<String> strings) {
89    List<String> quoted = new ArrayList<String>();
90    for (String s : strings) {
91        quoted.add("\"" + s.replace("\\", "\\\\")
92            .replace("\"", "\\\"").replace("\n", "\\n") + "\"");
93    }
94    return "[" + String.join(", ", quoted) + "]";
95}
